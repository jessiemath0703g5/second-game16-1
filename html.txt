        function genQLevel3() {
            let type = getNextType(5);
            if(type === 1) { 
                // 反比工程問題
                let p1, d1, totalWork;
                let possibleP2 = [];
                do {
                    p1 = rnd(3, 10);
                    d1 = rnd(4, 15);
                    totalWork = p1 * d1;
                    possibleP2 = [];
                    for(let i=2; i<=12; i++) {
                        if(totalWork % i === 0 && i !== p1) possibleP2.push(i);
                    }
                } while(possibleP2.length === 0);
                let p2 = possibleP2[Math.floor(Math.random() * possibleP2.length)];
                let d2 = totalWork / p2;
                let obj = rndItem(['機器人', '工人', '生產線']);
                return {
                    text: `有一份工程由 ${p1} 個${obj}同時工作，需要花 ${d1} 日才完成。如果工程趕工改由 ${p2} 個${obj}同時工作，幾日可以完成？`,
                    inputs: ['日'],
                    ans: [d2]
                };
            } else if (type === 2) { 
                // 間隔時間推算
                let intervalM = rndItem([15, 20, 25, 30, 40, 45]); 
                let startH = rnd(7, 10), startM = rnd(1, 5) * 10;
                let n = rnd(4, 10); 
                let totalM = startM + intervalM * (n - 1);
                let endH = startH + Math.floor(totalM / 60);
                let endM = totalM % 60;
                return {
                    text: `遊樂園接駁車每隔 ${intervalM} 分鐘發一班車，上午 ${startH} 時 ${startM} 分發出第 1 班車，第 ${n} 班車是上午幾時幾分發車？(請填入時、分)`,
                    inputs: ['時', '分'],
                    ans: [endH, endM]
                };
            } else if (type === 3) { 
                // 新舊機器相差
                let totalSec = rndItem([600, 720, 900, 1200, 1800, 2400]); 
                let divisors = [];
                for(let i=4; i<=40; i++) {
                    if(totalSec % i === 0) divisors.push(i);
                }
                let n1 = divisors[Math.floor(Math.random() * (divisors.length/2))];
                let n2 = divisors[Math.floor(Math.random() * (divisors.length/2) + divisors.length/2)];
                if(n1 === n2 || !n1 || !n2) { n1 = 10; n2 = 15; totalSec = 600; }
                if(n1 > n2) { let temp = n1; n1 = n2; n2 = temp; }
                
                let m = Math.floor(totalSec / 60), s = totalSec % 60;
                let diff = Math.abs((totalSec / n1) - (totalSec / n2));
                return {
                    text: `舊機器製作 ${n1} 個產品需要 ${m} 分鐘 ${s} 秒，改良後同樣時間可以製作 ${n2} 個。改良前後，製作「一個」產品的時間相差多少秒？`,
                    inputs: ['秒'],
                    ans: [diff]
                };
            } else if (type === 4) { 
                // 連續生產量跨中午
                let startH = rnd(9, 11), startM = rnd(1, 5) * 10;
                let mPerBatch = rndItem([12, 15, 18, 20]);
                let batches = rnd(5, 12);
                let k = rndItem([30, 40, 50, 60]);
                
                let totalM = startM + mPerBatch * batches;
                let endH = startH + Math.floor(totalM / 60);
                let endM = totalM % 60;

                return {
                    text: `水餃店煮一鍋水餃要 ${mPerBatch} 分鐘，一鍋有 ${k} 顆。從上午 ${startH} 時 ${startM} 分持續煮到 ${displayPeriod} ${displayEndH} 時 ${endM} 分，一共煮了幾顆水餃？`,
                    inputs: ['顆'],
                    ans: [batches * k]
                };
            } else { 
                // 休息時間累加

                let r = rnd(10, 30);
                let n = rnd(3, 5);
                let totalMin = (h * 60 + m) * n + r * (n - 1);
                // 陷阱題：問純分鐘或小時分

                        inputs: ['分'],

                        inputs: ['時', '分'],
                        ans: [Math.floor(totalMin / 60), totalMin % 60]
                    };
                }
            }
        }

        // === 檢查答案與回饋 ===
        function checkAnswer() {
            if(isPaused || isWaiting) return;

            let isCorrect = true;
            let userAns = [];
            for(let i=0; i<currentQuestion.ans.length; i++) {
                let val = parseInt(document.getElementById(`ans_${i}`).value);
                userAns.push(isNaN(val) ? 0 : val);
                if(val !== currentQuestion.ans[i] || isNaN(val)) {
                    isCorrect = false;
                }
            }

            totalAnswered++;
            if(isCorrect) correctCount++; else wrongCount++;
            
            // 紀錄歷史
            let userAnsStr = userAns.map((a, i) => `${a} ${currentQuestion.inputs[i]}`).join(', ');
            let correctAnsStr = currentQuestion.ans.map((a, i) => `${a} ${currentQuestion.inputs[i]}`).join(', ');
            
            roundHistory.push({
                qText: currentQuestion.text,
                userAns: userAnsStr,
                correctAns: correctAnsStr,
                isCorrect: isCorrect
            });

            isWaiting = true; 

            let fb = document.getElementById('feedbackText');
            fb.classList.remove('show-correct', 'show-wrong');
            void fb.offsetWidth; // trigger reflow

            if (isCorrect) {
                combo++;
                let points = 10 * (1 + (combo * 0.1));
                score += Math.floor(points);
                timeLeft += 5; 
                fb.innerHTML = "⭕ 正確！ +5秒";
                fb.classList.add('show-correct');
                triggerCheerleader(true);
                updateHUD();

                setTimeout(() => {
                    isWaiting = false;
                    nextQuestion();
                }, 1000); 
            } else {
                combo = 0;
                timeLeft -= 5;
                fb.innerHTML = `❌ 錯誤！ -5秒<br><span style="font-size: 2rem; color: #1F2937; text-shadow: none;">正確答案：${correctAnsStr}</span>`;
                fb.classList.add('show-wrong');
                triggerCheerleader(false);
                updateHUD();

                setTimeout(() => {
                    isWaiting = false;
                    nextQuestion();
                }, 2500); 
            }
        }

        // === 滿版白板功能 ===
        function initWhiteboard() {
            canvas = document.getElementById('whiteboard');
            ctx = canvas.getContext('2d');
            resizeCanvas();
            
            if(!isWhiteboardInitialized) {
                window.addEventListener('resize', resizeCanvas);

                canvas.addEventListener('mousedown', startDrawing);
                canvas.addEventListener('mousemove', draw);
                canvas.addEventListener('mouseup', stopDrawing);
                canvas.addEventListener('mouseout', stopDrawing);

                canvas.addEventListener('touchstart', handleTouchStart, {passive: false});
                canvas.addEventListener('touchmove', handleTouchMove, {passive: false});
                canvas.addEventListener('touchend', stopDrawing);
                
                isWhiteboardInitialized = true;
            }
        }

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            ctx.lineWidth = 3;
            ctx.lineCap = 'round';
            ctx.strokeStyle = '#4F46E5';
        }

        function toggleWhiteboard() {
            isWhiteboardActive = !isWhiteboardActive;
            canvas.classList.toggle('active');
            let btn = document.getElementById('btnWhiteboard');
            btn.innerText = isWhiteboardActive ? "👆 關閉白板 (作答)" : "✍️ 開啟白板";
            if(isWhiteboardActive) btn.classList.replace('secondary', 'danger');
            else btn.classList.replace('danger', 'secondary');
        }

        function clearWhiteboard() {
            if(ctx && canvas) ctx.clearRect(0, 0, canvas.width, canvas.height);
        }

        function getMousePos(evt) { return { x: evt.clientX, y: evt.clientY }; }
        function getTouchPos(evt) { return { x: evt.touches[0].clientX, y: evt.touches[0].clientY }; }

        function startDrawing(e) {
            if(!isWhiteboardActive) return;
            e.preventDefault(); // 修正：防止文字反藍和拖曳
            isDrawing = true;
            let pos = getMousePos(e);
            ctx.beginPath();
            ctx.moveTo(pos.x, pos.y);
        }
        function draw(e) {
            if(!isDrawing || !isWhiteboardActive) return;
            e.preventDefault(); // 修正：防止文字反藍和拖曳
            let pos = getMousePos(e);
            ctx.lineTo(pos.x, pos.y);
            ctx.stroke();
        }
        function handleTouchStart(e) {
            if(!isWhiteboardActive) return;
            e.preventDefault();
            isDrawing = true;
            let pos = getTouchPos(e);
            ctx.beginPath();
            ctx.moveTo(pos.x, pos.y);
        }
        function handleTouchMove(e) {
            if(!isDrawing || !isWhiteboardActive) return;
            e.preventDefault();
            let pos = getTouchPos(e);
            ctx.lineTo(pos.x, pos.y);
            ctx.stroke();
        }
        function stopDrawing() { isDrawing = false; }

                    <td style="font-size:1.2rem;">${item.isCorrect ? '✅' : '❌'}</td>
                </tr>`;
                revTbody.innerHTML += row;
            });
        }

        // === 排行榜與儲存 ===
        function saveScore() {
            if(totalAnswered === 0) return; // 沒作答不記錄
            
            let record = {
                name: player,
                level: `第${selLevel}關`,
                diff: selDiffName,
                score: score,
                total: totalAnswered,
                date: new Date().toLocaleString()
            };
            
            let lb = JSON.parse(localStorage.getItem('timeMathLB')) || [];
            lb.push(record);
            // 排序：先比分數(降冪)，同分比答題數(降冪)
            lb.sort((a, b) => b.score !== a.score ? b.score - a.score : b.total - a.total);
            lb = lb.slice(0, 10); // 取前10名
            localStorage.setItem('timeMathLB', JSON.stringify(lb));
        }

        function renderLeaderboard() {
            let lb = JSON.parse(localStorage.getItem('timeMathLB')) || [];
            
            let tbody1 = document.getElementById('lbBody');   // 結算頁的排行榜
            let tbody2 = document.getElementById('lbBody2');  // 選單頁的排行榜
            
            if(tbody1) tbody1.innerHTML = "";
            if(tbody2) tbody2.innerHTML = "";
            
            lb.forEach((r, idx) => {
                let rankText = "";
                if(idx === 0) rankText = "🥇 第一名";
                else if(idx === 1) rankText = "🥈 第二名";
                else if(idx === 2) rankText = "🥉 第三名";
                else rankText = `🏅 第${idx+1}名`;

                let tr = `<tr>
                    <td class="rank-icon">${rankText}</td>
                    <td><b>${r.name}</b></td>
                    <td>${r.level}</td>
                    <td>${r.diff}</td>
                    <td>${r.score} 分 / ${r.total} 題</td>
                    <td style="color:#6B7280; font-size:12px;">${r.date}</td>
                </tr>`;
                
                if(tbody1) tbody1.innerHTML += tr;
                if(tbody2) tbody2.innerHTML += tr;
            });
        }
